<?php
$n = 5;
for ($i = $n; $i >= 1; $i--) {
    for ($space = 0; $space < ($n - $i) * 3; $space++) {
        echo "&nbsp;";
    }
    for ($j = $i; $j >= 1; $j--) {
        echo $j . " ";
    }
    echo "<br>";
}
for ($i = 2; $i <= $n; $i++) {
    for ($space = 0; $space < ($n - 1) * 3; $space++) {
        echo "&nbsp;";
    }
    for ($j = 1; $j <= $i; $j++) {
        echo $j . " ";
    }
    echo "<br>";
}
?>
