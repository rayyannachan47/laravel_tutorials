2. Reverse Array

$array = [1, 2, 3, 6, 5];
$length = count($array);

for ($i = 0; $i < $length / 2; $i++) {
    $temp = $array[$i];
    $array[$i] = $array[$length - 1 - $i];
    $array[$length - 1 - $i] = $temp;
}

print_r($array);

3. Sort Array(Descending)

$array = [1, 2, 3, 4, 5];
$length = count($array);

for ($i = 0; $i < $length; $i++) {
    for ($j = 0; $j < $length - 1 - $i; $j++) {
        if ($array[$j] < $array[$j + 1]) {
            // Swap the elements
            $temp = $array[$j];
            $array[$j] = $array[$j + 1];
            $array[$j + 1] = $temp;
        }
    }
}

print_r($array);

13. Find a pair with the given difference

Input: arr[] = {5, 20, 3, 2, 50, 80}, n = 78
Output: Pair Found: (2, 80)

$arr = [5, 20, 3, 2, 50, 80,0];
$n = 78;
$pair = [];
foreach($arr as $val1){
	foreach($arr as $val2){
    	$sub = abs($val1 - $val2); 
        if($sub == $n){
        	$pair[] = "$val1,$val2";
        }
	}
}
print_r($pair);
